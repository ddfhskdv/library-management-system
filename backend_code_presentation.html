    @NotBlank(message = "Title is required")
    @Column(nullable = false, length = 200)
    private String title;

    @NotBlank(message = "Author is required")
    @Column(nullable = false, length = 150)

    @Column(nullable = false, length = 50)
    private String isbn;

    @Column(length = 100)
    private String category;

    @Column(length = 100)
    private String publisher;

    private Integer publicationYear;

    @Column(nullable = false)
    private Boolean available = true;

    private Integer totalCopies = 1;

    private Integer availableCopies = 1;

    public Book() {
    }

    public Book(String title, String author, String isbn, String category, Integer publicationYear) {
        this.title = title;
        this.author = author;
        this.isbn = isbn;
        this.category = category;
        this.publicationYear = publicationYear;
        this.available = true;
        this.totalCopies = 1;
        this.availableCopies = 1;

        this.availableCopies++;
        this.available = true;

StandardFineCalculator.java Fine Calculation & Strategy
◆ OOP Polymorphism Concrete implementation calculating late days * Rs. 5 as mandated by Step 10 return workflow. 
package com.college.library.service.impl;

import com.college.library.service.FineCalculator;
import org.springframework.stereotype.Service;

import java.time.LocalDate;
import java.time.temporal.ChronoUnit;

/**
 * Implements standard fine calculation rule specified in Step 10:
 * Fine = late days * ₹5
 */
@Service
public class StandardFineCalculator implements FineCalculator {

    private static final double DAILY_FINE_RATE = 5.0; // ₹5 per late day

    @Override
    public long calculateLateDays(LocalDate dueDate, LocalDate returnDate) {
        if (returnDate == null || dueDate == null || !returnDate.isAfter(dueDate)) {
            return 0;
        }
        return ChronoUnit.DAYS.between(dueDate, returnDate);
    }

    @Override
    public double calculateFine(LocalDate dueDate, LocalDate returnDate) {
        long lateDays = calculateLateDays(dueDate, returnDate);
        return lateDays * DAILY_FINE_RATE;
    }

    @Override
    public double getDailyRate() {
        return DAILY_FINE_RATE;
    }
}

package com.college.library.service.impl;

import com.college.library.entity.Book;
import com.college.library.exception.ResourceNotFoundException;
import com.college.library.repository.BookRepository;
import com.college.library.repository.TransactionRepository;
import com.college.library.service.BookService;
import org.springframework.stereotype.Service;
import org.springframework.transaction.annotation.Transactional;

import java.util.List;

/**
 * Implementation of BookService demonstrating:
 * - OOP Encapsulation & Polymorphism
 * - Java Collections & Stream API
 * - Exception Handling
 * - JPA Repository queries
 */
@Service
@Transactional
public class BookServiceImpl implements BookService {

    private final BookRepository bookRepository;
    private final TransactionRepository transactionRepository;

    public BookServiceImpl(BookRepository bookRepository, TransactionRepository transactionRepository) {
        this.bookRepository = bookRepository;
        this.transactionRepository = transactionRepository;
    }

    @Override
    @Transactional(readOnly = true)
    public List<Book> getAllBooks() {
        return bookRepository.findAll();
    }

    @Override
    @Transactional(readOnly = true)
    public Book getBookById(Long id) {
        return bookRepository.findById(id)
                .orElseThrow(() -> new ResourceNotFoundException("Book not found with ID: " + id));
    }

    @Override
    public Book addBook(Book book) {
        if (book.getTotalCopies() == null || book.getTotalCopies() <= 0) {
            book.setTotalCopies(1);
        }
        if (book.getAvailableCopies() == null) {
            book.setAvailableCopies(book.getTotalCopies());
        }
        book.setAvailable(book.getAvailableCopies() > 0);
        return bookRepository.save(book);
    }

    @Override
    public Book updateBook(Long id, Book updatedBook) {
        Book existing = getBookById(id);
        existing.setTitle(updatedBook.getTitle());
        existing.setAuthor(updatedBook.getAuthor());
        existing.setIsbn(updatedBook.getIsbn());
        existing.setCategory(updatedBook.getCategory());
        existing.setPublisher(updatedBook.getPublisher());
        existing.setPublicationYear(updatedBook.getPublicationYear());
        if (updatedBook.getTotalCopies() != null) {
            existing.setTotalCopies(updatedBook.getTotalCopies());
        }
        if (updatedBook.getAvailableCopies() != null) {
            existing.setAvailableCopies(updatedBook.getAvailableCopies());
        }
        existing.setAvailable(existing.getAvailableCopies() > 0);
        return bookRepository.save(existing);
    }

    @Override
    public void deleteBook(Long id) {
        Book book = getBookById(id);
        // Prevent deleting a book if it is currently issued
        boolean hasActiveIssues = transactionRepository.findFirstByBookIdAndStatus(id, "ISSUED").isPresent();
        if (hasActiveIssues) {
            throw new IllegalStateException("Cannot delete book '" + book.getTitle() + "' because it is currently issued to a member.");
        }
        bookRepository.delete(book);
    }

    @Override
    @Transactional(readOnly = true)
    public List<Book> searchBooks(String query) {
        if (query == null || query.trim().isEmpty()) {
            return bookRepository.findAll();
        }
        return bookRepository.searchBooks(query.trim());
    }

    @Override
    @Transactional(readOnly = true)
    public List<Book> getAvailableBooks() {
        return bookRepository.findByAvailable(true);
    }

    @Override
    @Transactional(readOnly = true)
    public boolean checkAvailability(Long id) {
        Book book = getBookById(id);
        return Boolean.TRUE.equals(book.getAvailable()) && book.getAvailableCopies() > 0;
    }
}

@Service
@Transactional
public class TransactionServiceImpl implements TransactionService {

    private final TransactionRepository transactionRepository;
    private final BookRepository bookRepository;
    private final MemberRepository memberRepository;
    private final FineCalculator fineCalculator;

    public TransactionServiceImpl(TransactionRepository transactionRepository,
                                  BookRepository bookRepository,
                                  MemberRepository memberRepository,
                                  FineCalculator fineCalculator) {
        this.transactionRepository = transactionRepository;
        this.bookRepository = bookRepository;
        this.memberRepository = memberRepository;
        this.fineCalculator = fineCalculator;
    }

    /**
     * Step 10: Issue Book Logic Flow:
     * Select member -> Select available book -> Issue book -> available = false -> Create transaction -> Due date = issue date + 14 days
     */
    @Override
    public Transaction issueBook(IssueRequest request) {
        // 1. Select member
        Member member = memberRepository.findById(request.getMemberId())
                .orElseThrow(() -> new ResourceNotFoundException("Member not found with ID: " + request.getMemberId()));

        if (Boolean.FALSE.equals(member.getActive())) {
            throw new IllegalStateException("Member account is currently inactive.");
        }

        // 2. Select available book
        Book book = bookRepository.findById(request.getBookId())
                .orElseThrow(() -> new ResourceNotFoundException("Book not found with ID: " + request.getBookId()));

        // Check availability
        if (Boolean.FALSE.equals(book.getAvailable()) || book.getAvailableCopies() == null || book.getAvailableCopies() <= 0) {
            throw new BookNotAvailableException("Book '" + book.getTitle() + "' is not available for issue.");
        }

        // 3. Issue book & update available
        book.markIssued();
        bookRepository.save(book);

        // 4. Create transaction
        int days = (request.getLoanDays() != null && request.getLoanDays() > 0) ? request.getLoanDays() : 14;
        LocalDate issueDate = LocalDate.now();
        LocalDate dueDate = issueDate.plusDays(days);

        Transaction transaction = new Transaction(book, member, issueDate, dueDate);
        return transactionRepository.save(transaction);
    }

    /**
     * Step 10: Return Book Logic Flow:
     * Return book -> Calculate late days -> Fine = late days * ₹5 -> Mark transaction returned -> available = true
     */
    @Override
    public ReturnResponse returnBook(ReturnRequest request) {
        // 1. Retrieve transaction
        Transaction transaction = transactionRepository.findById(request.getTransactionId())
                .orElseThrow(() -> new ResourceNotFoundException("Transaction not found with ID: " + request.getTransactionId()));

        if ("RETURNED".equalsIgnoreCase(transaction.getStatus())) {
            throw new IllegalStateException("This transaction has already been completed and returned.");
        }

        LocalDate returnDate = request.getReturnDate() != null ? request.getReturnDate() : LocalDate.now();
        transaction.setReturnDate(returnDate);

        // 2. Calculate late days and fine (Fine = late days * ₹5)
        long lateDays = fineCalculator.calculateLateDays(transaction.getDueDate(), returnDate);
        double fineAmount = fineCalculator.calculateFine(transaction.getDueDate(), returnDate);

        // 3. Mark transaction returned
        transaction.setStatus("RETURNED");
        transaction.setFineAmount(fineAmount);
        transactionRepository.save(transaction);

        // 4. Mark available = true & increment copies
        Book book = transaction.getBook();
        book.markReturned();
        bookRepository.save(book);

        String message = lateDays > 0
                ? String.format("Book returned late by %d day(s). Late fine of ₹%.2f applied.", lateDays, fineAmount)
                : "Book returned successfully on time. No fine applied.";

        return new ReturnResponse(
                transaction.getId(),
                book.getId(),
                book.getTitle(),
                transaction.getMember().getId(),
                transaction.getMember().getName(),
                transaction.getIssueDate(),
                transaction.getDueDate(),
                returnDate,
                lateDays,
                fineCalculator.getDailyRate(),
                fineAmount,
                message
        );
    }

    @Override
    @Transactional(readOnly = true)
    public List<Transaction> getAllTransactions() {
        return transactionRepository.findAllByOrderByCreatedAtDesc();
    }

    @Override
    @Transactional(readOnly = true)
    public List<Transaction> getActiveTransactions() {
        return transactionRepository.findByStatusOrderByIssueDateDesc("ISSUED");
    }

    @Override
    @Transactional(readOnly = true)
    public List<Transaction> getOverdueTransactions() {
        return transactionRepository.findOverdueTransactions(LocalDate.now());
    }

    public Transaction getTransactionById(Long id) {
        return transactionRepository.findById(id)
                .orElseThrow(() -> new ResourceNotFoundException("Transaction not found with ID: " + id));
    }

 */
@Service
@Transactional(readOnly = true)
public class DashboardServiceImpl implements DashboardService {

    private final BookRepository bookRepository;
    private final MemberRepository memberRepository;
    private final TransactionRepository transactionRepository;

    public DashboardServiceImpl(BookRepository bookRepository,
                                MemberRepository memberRepository,

        this.memberRepository = memberRepository;
        this.transactionRepository = transactionRepository;
    }

    @Override
    public DashboardStats getDashboardStatistics() {

        // Calculate total book copies dynamically using Java Streams
        long totalCopies = books.stream()
                .mapToLong(b -> b.getTotalCopies() != null ? b.getTotalCopies() : 1)
                .sum();

        long availableCopies = books.stream()
                .mapToLong(b -> b.getAvailableCopies() != null ? b.getAvailableCopies() : (Boolean.TRUE.equals(b.getAvailable()) ? 1 : 0))
                .sum();

        long issuedCount = transactionRepository.countByStatus("ISSUED");
        long totalMembers = memberRepository.count();
        long overdueCount = transactionRepository.countOverdueTransactions(LocalDate.now());
        Double totalFines = transactionRepository.sumTotalFines();

        return new DashboardStats(
                totalCopies,
                availableCopies,
                issuedCount,
                totalMembers,
                overdueCount,
                totalFines != null ? totalFines : 0.0
        );

Presentation & Viva Voce Q&A Cheat Sheet Frequently Asked Examiner Questions
Q1: Where is Object-Oriented Inheritance demonstrated in your codebase?
Answer: Inheritance is demonstrated in two key places:
1. BaseEntity.java is an abstract @MappedSuperclass that provides shared primary key (id) and audit timestamps (createdAt, updatedAt) to all database models.
2. Person.java is an abstract superclass providing common identity fields (name, email, phone), which is inherited by Member.java.
Q2: How is the fine calculation implemented according to the Step 10 specification?
Answer: We designed a decoupled FineCalculator interface implemented by StandardFineCalculator.java. When a book is returned, it calculates:
lateDays = ChronoUnit.DAYS.between(dueDate, returnDate)
If lateDays > 0, the fine is calculated as: fine = lateDays * 5.0 (₹5 per late day). If returned on or before the due date, fine is ₹0.0.
Q3: Are the Step 11 Dashboard numbers hard-coded or dynamic?
Answer: They are 100% dynamic! DashboardServiceImpl.java queries the repositories and uses the Java Stream API to compute total copies, available copies, active issued loans, and registered members. DataInitializer.java seeds realistic initial data (100 total book copies, 75 available, 25 issued, 50 members) so that the live numbers exactly match the project prompt during presentation.
Q4: How does your project handle database operations and what database is used?
Answer: We use Spring Data JPA with Hibernate as the ORM. The default database is H2 In-Memory Database with the web console enabled at http://localhost:8080/h2-console. This requires zero external software setup on any machine. For production, application.properties also contains pre-configured MySQL connection settings.
Q5: How are errors and exceptions handled in your REST APIs?
Answer: Centralized exception handling is implemented using @RestControllerAdvice in GlobalExceptionHandler.java. Custom business exceptions like ResourceNotFoundException and BookNotAvailableException return clean, structured JSON error responses with proper HTTP status codes (404, 400).
Q6: How do you run the complete project?
Answer: From the project root folder, execute mvn spring-boot:run or double-click run.bat. Spring Boot starts the embedded Tomcat server on port 8080 and serves both the REST API backend and the responsive Single Page Web UI together at http://localhost:8080.
